// from src/contracts/repo.ts
export interface Repo {
  /** Catalog + custom exercises (one list). Catalog rows are read-only: update/remove throw READONLY_CATALOG. */
  exercises: EntityRepo<"exercise"> & {
    search(query: ExerciseSearch): Promise<ExerciseRow[]>;
    bySlug(slug: string): Promise<ExerciseRow | undefined>;
    /** Exercises most recently logged, newest first. */
    recent(limit?: number): Promise<ExerciseRow[]>;
  };
  /** Starred exercises (catalog or custom); synced offline-first. The row id is favoriteId(userId, exerciseId). */
  favorites: {
    /** Exercises the user starred (alive favorites whose exercise still exists), in position order (new stars append). */
    list(): Promise<ExerciseRow[]>;
    /** Ids of the starred exercises. */
    ids(): Promise<Set<string>>;
    isFavorite(exerciseId: string): Promise<boolean>;
    /** Star or unstar (a tombstone; starring again restores it with a newer HLC). Resolves to whether it is a favorite now. */
    toggle(exerciseId: string): Promise<boolean>;
  };
  machineProfiles: EntityRepo<"machineProfile"> & {
    forExercise(exerciseId: string): Promise<MachineProfileRow[]>;
    /** The profile most recently used for this exercise (drives the "default to last used" machine picker). */
    lastUsedFor(exerciseId: string): Promise<MachineProfileRow | undefined>;
  };
  bodyMetrics: EntityRepo<"bodyMetric"> & {
    /** Inclusive, ordered by measuredAt ascending. */
    inRange(from: LocalDateString, to: LocalDateString): Promise<BodyMetricRow[]>;
    latest(): Promise<BodyMetricRow | undefined>;
  };
  templates: EntityRepo<"template"> & {
    exercisesOf(templateId: string): Promise<TemplateExerciseRow[]>;
  };
  templateExercises: EntityRepo<"templateExercise">;
  workouts: EntityRepo<"workout"> & {
    /** The unfinished workout (endedAt null, not deleted), newest startedAt first if several. */
    active(): Promise<WorkoutRow | undefined>;
    /** By localDate, inclusive, newest first. */
    inRange(from: LocalDateString, to: LocalDateString): Promise<WorkoutRow[]>;
    /** Sets endedAt (default: now). */
    finish(id: string, endedAt?: IsoDateTime): Promise<WorkoutRow>;
  };
  workoutExercises: EntityRepo<"workoutExercise"> & {
    /** Ordered by position. */
    forWorkout(workoutId: string): Promise<WorkoutExerciseRow[]>;
  };
  sets: EntityRepo<"workoutSet"> & {
    /** Ordered by position. */
    forWorkoutExercise(workoutExerciseId: string): Promise<WorkoutSetRow[]>;
    forWorkout(workoutId: string): Promise<WorkoutSetRow[]>;
    /** Every live (non-deleted) set of one progression series, any workout, unordered. Index: workoutSet.seriesKey. */
    forSeries(seriesKey: string): Promise<WorkoutSetRow[]>;
    lastTime(args: {
      exerciseId: string;
      machineProfileId: string | null;
      /** Only sessions strictly before this localDate (default: today). */
      beforeLocalDate?: LocalDateString;
    }): Promise<LastTimeEntry | undefined>;
  };
  catalog: {
    /** CatalogMeta.version of the cached catalog, or null if never fetched. */
    version(): Promise<number | null>;
    /** Replace the cached catalog rows (called by the sync engine after a 200 from /api/catalog). */
    replace(catalog: CatalogResponse): Promise<void>;
  };
  /** Run several writes atomically (one Dexie transaction, one outbox batch). Nested calls join the outer one. */
  transact<T>(fn: (tx: Repo) => Promise<T>): Promise<T>;
}

## Deliverables

- **Start a workout** empty or from a template (`StartWorkoutCard` exported from `src/components/workout/StartWorkoutCard.tsx` for Wave 3's `/today`: props only, no data fetching; it receives `activeWorkout`, `templates`, and callbacks). Creating a workout writes `localDate` from the device's local date (`toLocalDate(now, Intl timezone)`) and `timezone`.
- **Exercise picker** split by modality (tabs Free weight / Machine / Cable / Bodyweight), search, recent, built from `repo.exercises.search` and stream C's presentational components. MACHINE and CABLE exercises **ask for a machine profile** (`MachineProfile`), defaulting to `repo.machineProfiles.lastUsedFor(exerciseId)`; creating a profile inline is possible.
- **Set rows** (`SetRow`): labelled by `LoadType` (per hand, per side/plates, stack, +BW, total) and `TrackingType`; planned/active/completed/PR/warm-up (muted) states; "last time" values as ghost text from `repo.sets.lastTime`; weight entered in the user's unit (`inputUnit` stored), converted with `toKg`; steppers use `defaultIncrement`; RIR/RPE entry per the user's `effortScale` (store RIR); set types (WARMUP/WORKING/DROP/AMRAP); completing a set stamps `completedAt`. All writes go through `repo` (offline).
- **Rest timer** (`RestTimer`, inline + sheet): counts down to an **absolute `restEndsAt`** persisted locally (survives backgrounding; iOS pauses timers), shows done/paused, uses the Screen Wake Lock API while a workout is active (feature-detected), haptic/ding where supported.
- **Live PR toast**: after completing a set, `detectNewPrs(history, newSet)` shows a toast (stream B's `Toast`); sync-confirmed PRs from `useOnSynced`/`newPrs` are shown once.
- **Supersets and reordering** (`supersetKey`, fractional-index `position`), add/remove exercises and sets mid-workout, **finish summary** (duration, sets, volume, PRs), discard workout. Call `setWorkoutActive(true)` while a workout is active and `false` on finish/discard.
- **Templates** CRUD (`(app)/templates`, `components/templates/**`): folders, ordering, per-exercise target sets/reps/RIR/rest/machine profile, "save workout as template".
- **History** (`(app)/history`, detail page): list by `localDate`, edit/delete past workouts and sets (all through `repo`).
- **Custom exercises**: `(app)/exercises/new` and `(app)/exercises/[id]/edit` (modality/equipment/loadType/tracking/muscles; the muscles ride in the Exercise mutation) and **machine profiles** `(app)/settings/machines` (name, gym, brand, label unit, pin step, base load, linked exercise).
- Styles only in `src/styles/workout.css`. The workout screens are the gym UI: large touch targets (>= 44px), dark by default once installed, no photo backdrop, one-handed layout, no layout shift when numbers change (use `num`).

**Notes.** Machine and free-weight progress stay separate: a set's `machineProfileId` determines its series; never offer to "convert" between machines. iOS home-screen apps have separate cookie storage from Safari and kill timers in the background: use absolute timestamps everywhere.

// from src/contracts/sync.ts
export type SyncRequest = z.infer<typeof syncRequestSchema>;

export interface MutationResult {
  mutationId: string;
  /** applied: changed the row. noop: valid, already merged (idempotent replay). rejected: refused and dead-lettered. */
  status: "applied" | "noop" | "rejected";
  reason?: RejectReason;
  message?: string;
}

export interface SyncResponse {
  /** One entry per request mutation, same order. A rejected mutation never blocks the rest. */
  results: MutationResult[];
  changes: SyncChanges;
  /** PRs rebuilt for the series touched by this request (drives the "new PR" toast after sync). */
  newPrs: PersonalRecordRow[];
  /** Send this as `cursor` next time. When hasMore, it is the version of the last returned row. */
  cursor: number;
  /** true = more pages are waiting: call again immediately with `cursor` and no mutations. */
  hasMore: boolean;
  /**
   * true = request.cursor was older than the server's gcFloorVersion (tombstones were collected): `changes` is empty;
   * the client must wipe its local synced tables (keeping the outbox) and restart from cursor 0.
   */
  reset: boolean;
  serverTime: IsoDateTime;
  serverHlc: Hlc;
}

export interface SyncStatus {
  state: SyncState;
  /** Outbox mutations not yet acknowledged by the server. */
  pendingCount: number;
  lastSyncedAt: IsoDateTime | null;
  lastError: string | null;
  /** Mutations the server rejected in the last sync (dead-lettered). */
  rejectedCount: number;
}

export interface SyncResult {
  ok: boolean;
  pushed: number;
  pulled: number;
  rejected: number;
  newPrs: PersonalRecordRow[];
  /** Set when ok is false. */
  error?: string;
  /** true when another sync was already running (navigator.locks) and this call did nothing. */
  skipped?: boolean;
}

/** Info passed to useOnSynced listeners after every completed sync that pulled or pushed something. */
export type SyncedInfo = Pick<SyncResult, "pushed" | "pulled" | "rejected" | "newPrs">;

/** Lexicographic HLC compare (valid because of the fixed-width format). "" sorts first. */
export function compareHlc(a: Hlc, b: Hlc): number;

export function isHlc(value: string): boolean;

export function emptyChanges(): SyncChanges;
```

// from src/contracts/repo.ts
export interface EntityRepo<E extends EntityName> {
  get(id: string): Promise<EntityTypeMap[E]["row"] | undefined>;
  /** Live (non-deleted) rows, unordered unless a more specific method says otherwise. */
  list(): Promise<EntityTypeMap[E]["row"][]>;
  create(input: EntityTypeMap[E]["create"], opts?: { id?: string }): Promise<EntityTypeMap[E]["row"]>;
  update(id: string, patch: EntityTypeMap[E]["patch"]): Promise<EntityTypeMap[E]["row"]>;
  /** Soft delete (tombstone). Idempotent. */
  remove(id: string): Promise<void>;
}

export interface Repo {
  /** Catalog + custom exercises (one list). Catalog rows are read-only: update/remove throw READONLY_CATALOG. */
  exercises: EntityRepo<"exercise"> & {
    search(query: ExerciseSearch): Promise<ExerciseRow[]>;
    bySlug(slug: string): Promise<ExerciseRow | undefined>;
    /** Exercises most recently logged, newest first. */
    recent(limit?: number): Promise<ExerciseRow[]>;
  };
  machineProfiles: EntityRepo<"machineProfile"> & {
    forExercise(exerciseId: string): Promise<MachineProfileRow[]>;
    /** The profile most recently used for this exercise (drives the "default to last used" machine picker). */
    lastUsedFor(exerciseId: string): Promise<MachineProfileRow | undefined>;
  };
  bodyMetrics: EntityRepo<"bodyMetric"> & {
    /** Inclusive, ordered by measuredAt ascending. */
    inRange(from: LocalDateString, to: LocalDateString): Promise<BodyMetricRow[]>;
    latest(): Promise<BodyMetricRow | undefined>;
  };
  templates: EntityRepo<"template"> & {
    exercisesOf(templateId: string): Promise<TemplateExerciseRow[]>;
  };
  templateExercises: EntityRepo<"templateExercise">;
  workouts: EntityRepo<"workout"> & {
    /** The unfinished workout (endedAt null, not deleted), newest startedAt first if several. */
    active(): Promise<WorkoutRow | undefined>;
    /** By localDate, inclusive, newest first. */
    inRange(from: LocalDateString, to: LocalDateString): Promise<WorkoutRow[]>;
    /** Sets endedAt (default: now). */
    finish(id: string, endedAt?: IsoDateTime): Promise<WorkoutRow>;
  };
  workoutExercises: EntityRepo<"workoutExercise"> & {
    /** Ordered by position. */
    forWorkout(workoutId: string): Promise<WorkoutExerciseRow[]>;
  };
  sets: EntityRepo<"workoutSet"> & {
    /** Ordered by position. */
    forWorkoutExercise(workoutExerciseId: string): Promise<WorkoutSetRow[]>;
    forWorkout(workoutId: string): Promise<WorkoutSetRow[]>;
    /** Every live (non-deleted) set of one progression series, any workout, unordered. Index: workoutSet.seriesKey. */
    forSeries(seriesKey: string): Promise<WorkoutSetRow[]>;
    lastTime(args: {
      exerciseId: string;
      machineProfileId: string | null;
      /** Only sessions strictly before this localDate (default: today). */
      beforeLocalDate?: LocalDateString;
    }): Promise<LastTimeEntry | undefined>;
  };
  catalog: {
    /** CatalogMeta.version of the cached catalog, or null if never fetched. */
    version(): Promise<number | null>;
    /** Replace the cached catalog rows (called by the sync engine after a 200 from /api/catalog). */
    replace(catalog: CatalogResponse): Promise<void>;
  };
  /** Run several writes atomically (one Dexie transaction, one outbox batch). Nested calls join the outer one. */
  transact<T>(fn: (tx: Repo) => Promise<T>): Promise<T>;
}

/** Hook signature (src/client/repo): re-runs `query` whenever the Dexie tables it reads change (wraps dexie-react-hooks). */
export type UseLive = <T>(query: () => Promise<T> | T, deps?: readonly unknown[]) => T | undefined;

// from src/contracts/prs.ts
/** The columns PR logic reads from a set (works on server rows and Dexie rows alike). */
export interface PrSetInput {
  id: string;
  workoutId: string;
  exerciseId: string;
  machineProfileId: string | null;
  seriesKey: string;
  setType: SetType;
  /** null = planned set: never counts. */
  completedAt: IsoDateTime | null;
  reps: number | null;
  effectiveLoadKg: number | null;
  e1rmKg: number | null;
}

/** The current best value of one PR type for one series. (type, reps) is unique per series; reps = 0 unless REP_MAX. */
export interface PrCandidate {
  seriesKey: string;
  exerciseId: string;
  machineProfileId: string | null;
  type: PrType;
  reps: number;
  value: number;
  setId: string | null;
  workoutId: string;
  achievedAt: IsoDateTime;
}

/**
 * Current PRs for ONE series from its completed sets. WARMUP and DROP sets never count; planned sets (completedAt null) never count.
 *  E1RM: best e1rmKg.  MAX_LOAD: best effectiveLoadKg (reps >= 1).  REP_MAX: best effectiveLoadKg for exactly N reps (one row per N).
 *  MAX_REPS: most reps in a set (bodyweight/no-load work).  MAX_SET_VOLUME: best effectiveLoadKg x reps.
 *  MAX_SESSION_VOLUME: best per-workout sum of effectiveLoadKg x reps.
 * Ties keep the EARLIEST achievement. Machine series never merge: the caller passes one seriesKey at a time.
 */
export type ComputeSeriesPrs = (sets: PrSetInput[]) => PrCandidate[];

/** PRs that `newSet` would set against `history` (strictly better than the prior best): drives the live PR toast. */
export type DetectNewPrs = (history: PrSetInput[], newSet: PrSetInput) => PrCandidate[];
```

## Deliverables

- `src/lib/sync/hlc.ts` (pure): clock with `now/tick/receive` producing the frozen HLC string format, clamping a remote HLC more than `SYNC_MAX_CLOCK_SKEW_MS` ahead of local wall time; `src/lib/sync/merge.ts` (pure, **shared by client and server**): per-field last-write-wins by HLC, commutative, idempotent, `deletedAt` is an ordinary LWW field; **property test (fast-check)** that two devices applying the same mutations in any order and with duplicates converge to identical rows.
- `src/lib/sync/prs.ts`: `computeSeriesPrs` and `detectNewPrs` (WARMUP/DROP/planned never count; E1RM, MAX_LOAD, REP_MAX per rep count, MAX_REPS, MAX_SET_VOLUME, MAX_SESSION_VOLUME; ties keep the earliest). Unit + property tests.
- `src/server/sync/apply.ts` `applyServerMutations(ctx, request)`: Zod-validate each mutation independently (a bad one -> `rejected` + `SyncDeadLetter`, never blocks the rest); check ownership (`NOT_OWNER` for ids owned by another user, `CATALOG_READONLY` for global exercises) and references (`ENTITY_REFS`, `MISSING_REFERENCE`); inside ONE short transaction (15 s timeout is configured on the client) lock the user's `SyncState` row (`SELECT ... FOR UPDATE` via `$queryRaw`), merge fields per HLC with the shared `merge.ts` (clamp client HLCs to server time + 5 min), assign each changed row the next `version`, upsert rows (Decimal conversion here; `muscles` -> `ExerciseMuscle` rows), tombstone orphaned children of tombstoned parents, derive `workoutId/exerciseId/machineProfileId/seriesKey/localDate/effectiveLoadKg/e1rmKg` for sets (bodyweight for BODYWEIGHT_PLUS from the workout's `bodyweightKg`, else the latest `BodyMetric`), rebuild PRs for each affected `seriesKey` (`computeSeriesPrs`, `PersonalRecord` upserts, `previousValue`), then return the pull page (`version > cursor`, <= 1000 rows ordered by version across entities, `hasMore`, `cursor`) and `newPrs`. Replaying a batch is harmless (`noop`). `reset: true` when `cursor < gcFloorVersion`.
- `src/server/sync/gc.ts` + `GET /api/sync/gc` (`src/app/api/sync/gc/route.ts`, Bearer `CRON_SECRET`; scheduled daily in `vercel.json`): hard-delete tombstones older than `TOMBSTONE_TTL_DAYS`, children before parents (FKs are `NoAction`), then raise `gcFloorVersion`. `POST /api/sync` (`src/app/api/sync/route.ts`): `withUserRoute` -> `syncRequestSchema.safeParse` (400 on failure) -> `applyServerMutations`.
- `src/client/db/**`: Dexie schema in a per-user database `overload-<userId>` (tables for the 8 entities + catalog + `outbox` + `meta` with cursor/HLC/deviceId), versioned migrations, no cross-user leakage.
- `src/client/repo/**`: implement the frozen `Repo` interface exactly: every write is ONE Dexie transaction (row + per-field HLC + outbox mutation); strict-schema validation on create/update (forbidden fields throw `RepoError`); soft deletes; local mirror of derived set columns; `lastTime`, `lastUsedFor`, `active`, `search`, `recent`, `transact`, `catalog.replace`; `useLive` wrapping `useLiveQuery`. Export `repo` and `useLive` from `src/client/repo/index.ts`.
- `src/client/sync/**`: the engine (push up to 200 outbox mutations, apply the response transactionally, merge pulled rows with local rows, drop acknowledged outbox entries, `reset` handling, `hasMore` paging), triggers (app start, `online`, `visibilitychange`, 1.5 s after a write, every 30 s while pending, manual `syncNow`), `navigator.locks` single-flight, status store behind `useSyncStatus`, `useOnSynced`, catalog refresh with ETag (`/api/catalog`, 304-aware) feeding `repo.catalog.replace`, signed-out handling (keep pending changes, status `signed-out`: "Sign in to sync N changes"), `SyncProvider` that opens the user's database and starts/stops the engine. Replace the Phase 0 no-op stubs; keep names and signatures.
- Tests: **merge convergence property test**; **integration tests on a Neon branch** (replaying a batch is harmless; writes to another user's ids are `rejected`; orphaned children are tombstoned; concurrent pushes never skip rows (run two pushes in parallel and assert the union is pulled with strictly increasing versions); GC order and `reset`); **fake-indexeddb client tests** (repo semantics, outbox drain, `reset`, per-user isolation); `e2e/sync.spec.ts` (two browser contexts, same user, converge both ways; an offline edit syncs after going online). **Notes.** iOS has no Background Sync: the triggers replace it. Server Action ids change on every deploy, which is why sync is a route handler. Keep transactions short (Neon pooled connections). Reads for the pull page may run outside the write transaction but must be consistent with the cursor you return (never return a cursor beyond rows you included). Never trust a client-sent `userId`, `version`, `hlc` of another entity, or derived column.
